
#! /usr/bin/python
from scapy.all import *
rang = '192.168.43.1' # IP de départ pour ICMP
ip0=rang.split('.')
for ip in range(1,9): # Plage IP à partir de rang
    adr=ip0[0]+'.'+ip0[1]+'.'+ ip0[2]+'.'+str(ip)
    print(adr)
    rep, non_rep = sr(IP(dst=adr) / ICMP() , timeout=2 ) 
    print(rep)
    for elem in rep : # elem représente un couple (paquet émis,paquet reçu)
        if elem[1].type == 0 : # 0 <=> echo-reply
            print (elem[1].src + ' a renvoye un echo-reply ')
